import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';

/**

 * @author Jacopo Ricciardi

 * @function HomeScreen
 * @description Schermata home dell'app

 * @param {Function} props.onPlay - Callback chiamata quando il giocatore avvia la partita
 * @returns {JSX.Element} Schermata home renderizzata
 */
export default function HomeScreen({ onPlay }) {
  return (
    <View style={stileHome.container}>

}

});
